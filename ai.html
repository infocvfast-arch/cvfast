export default async function handler(req, res) { res.setHeader("Access-Control-Allow-Origin", "*"); res.setHeader("Access-Control-Allow-Methods", "POST, OPTIONS"); res.setHeader("Access-Control-Allow-Headers", "Content-Type"); if (req.method === "OPTIONS") { return res.status(200).end(); } if (req.method !== "POST") { return res.status(405).json({ success: false, error: "Method not allowed" }); } try { const { fullName, email, phone, location, targetRole, summary, experience, education, skills, languages, paymentOrderID } = req.body || {}; if (!fullName || !targetRole) { return res.status(400).json({ success: false, error: "Full name and target role are required." }); } if (!paymentOrderID) { return res.status(402).json({ success: false, error: "Verified payment is required." }); } if (!process.env.OPENAI_API_KEY) { return res.status(500).json({ success: false, error: "OPENAI_API_KEY is not configured." }); } const prompt = ` Create a professional ATS-friendly CV for the Dutch and international job market. CANDIDATE Full name: ${fullName} Email: ${email || ""} Phone: ${phone || ""} Location: ${location || ""} Target role: ${targetRole} Professional background: ${summary || ""} Work experience: ${experience || ""} Education: ${education || ""} Skills: ${skills || ""} Languages: ${languages || ""} INSTRUCTIONS Write the CV in professional English. Use this structure where relevant: FULL NAME Target Role PROFESSIONAL SUMMARY CORE SKILLS WORK EXPERIENCE EDUCATION LANGUAGES ADDITIONAL INFORMATION Rules: - Optimize for ATS systems. - Use clear professional wording. - Focus the content on the target role. - Improve grammar, clarity, structure and impact. - Use concise bullet points for work experience. - Emphasize achievements only when supported by the supplied information. - Never invent companies. - Never invent job titles. - Never invent employment dates. - Never invent education. - Never invent qualifications. - Never invent languages. - Never invent statistics, percentages, revenue numbers or achievements. - Preserve factual information supplied by the candidate. - Do not mention AI. - Do not include commentary before or after the CV. - Return only the finished CV. `; const response = await fetch( "https://api.openai.com/v1/responses", { method: "POST", headers: { Authorization: `Bearer ${process.env.OPENAI_API_KEY}`, "Content-Type": "application/json" }, body: JSON.stringify({ model: "gpt-5-mini", input: prompt }) } ); const data = await response.json(); if (!response.ok) { console.error("OpenAI error:", data); return res.status(response.status).json({ success: false, error: data?.error?.message || "OpenAI CV generation failed." }); } let cv = ""; if (typeof data.output_text === "string") { cv = data.output_text; } if (!cv && Array.isArray(data.output)) { for (const item of data.output) { if (!Array.isArray(item.content)) continue; for (const content of item.content) { if ( content.type === "output_text" && typeof content.text === "string" ) { cv += content.text; } } } } cv = cv.trim(); if (!cv) { return res.status(500).json({ success: false, error: "OpenAI returned an empty CV." }); } return res.status(200).json({ success: true, cv }); } catch (error) { console.error("generate-cv error:", error); return res.status(500).json({ success: false, error: "Unable to generate CV." }); } }
